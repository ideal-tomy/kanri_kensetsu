### 1.5 PDF生成エンジン

`puppeteer` または `playwright` で HTML → PDF。

```typescript
// src/lib/report/pdf-generator.ts
import { chromium } from 'playwright';
import Handlebars from 'handlebars';

export async function generateReportPDF(
  template: ReportTemplate,
  data: Record<string, any>
): Promise<Buffer> {
  // 1. レイアウトHTMLにデータを埋め込み
  const compiled = Handlebars.compile(template.layout_html);
  const html = compiled(data);
  
  // 2. ヘッドレスブラウザでPDF生成
  const browser = await chromium.launch();
  const page = await browser.newPage();
  await page.setContent(html, { waitUntil: 'networkidle' });
  const pdf = await page.pdf({
    format: 'A4',
    printBackground: true,
    margin: { top: '20mm', bottom: '20mm', left: '15mm', right: '15mm' },
  });
  await browser.close();
  
  return pdf;
}
```

## 2. トーク機能

### 2.1 設計方針

> 現場ポケットで実現している機能そのままのレベルでOK。LINEなど多くの人が使用しているもののUIUXを使って直観的に。低コスト運用前提。

**核心**：LINE風UIで誰でも使える。ただしログ保管は最小限でコスト抑制。

### 2.2 ログ保管ポリシー（コスト最適化）

```
┌──────────────────────────────────────────────────┐
│ ログ保管の3階層                                     │
├──────────────────────────────────────────────────┤
│ HOT   (0〜30日)  Supabase DB に保管、全文検索可    │
│ WARM  (31〜90日) Supabase Storage にJSON圧縮       │
│ COLD  (91日〜)   削除 or 別ストレージへ移動         │
└──────────────────────────────────────────────────┘

設定可能（管理画面で変更可能）:
- 保管期間：30日 / 90日 / 1年 / 無制限
- ピン留めメッセージは保管期間関係なく永続保存
- 重要マーク付きメッセージも永続保存
```

```typescript
// src/lib/messages/retention.ts
// Supabase Edge Functions で日次実行（Cron）

export async function archiveOldMessages() {
  const supabase = createServerClient();
  
  // 各会社の保管ポリシーを取得
  const { data: policies } = await supabase
    .from('message_retention_policies')
    .select('*');
  
  for (const policy of policies ?? []) {
    const hotDate = subDays(new Date(), policy.hot_days);
    const warmDate = subDays(new Date(), policy.warm_days);
    
    // HOT → WARM 移動（ピン留め・重要は除外）
    const { data: toArchive } = await supabase
      .from('messages')
      .select('*')
      .lt('created_at', hotDate.toISOString())
      .eq('is_pinned', false)
      .eq('is_important', false);
    
    if (toArchive && toArchive.length > 0) {
      // JSON圧縮してStorageへ
      const yearMonth = format(hotDate, 'yyyy-MM');
      const filePath = `archives/${policy.company_id}/${yearMonth}.json.gz`;
      const compressed = gzipSync(JSON.stringify(toArchive));
      
      await supabase.storage
        .from('message-archives')
        .upload(filePath, compressed, { contentType: 'application/gzip' });
      
      // DBから削除
      await supabase
        .from('messages')
        .delete()
        .in('id', toArchive.map(m => m.id));
```

## 3. 工程管理（半自動連動）

### 3.1 設計方針

> 工程管理の初期情報（工期、作業内容、スケジュール）は管理者・監督が初期入力。  
> 現場作業員の進捗報告（画像など）が自動（半自動）で反映。  
> 【現場の作業報告→報告書に反映→報告書の内容や現場の作業報告から全体の進捗が進んでいく】

**核心**：管理者の初期計画 + 現場の実績入力 + 自動集計の3層構造。

### 3.2 アーキテクチャ

```
┌──────────────────────────────────────────────────┐
│ 工程管理の連動フロー                                │
└──────────────────────────────────────────────────┘

【初期登録】管理者・監督
  ↓
  ┌──────────────────┐
  │ project_phases   │  ← 工程の大区分（基礎/躯体/仕上げ）
  │ - phase_name     │
  │ - planned_start  │
  │ - planned_end    │
  │ - target_qty     │
  └─────────┬────────┘
            ↓ 紐づく
  ┌──────────────────┐
  │ tasks            │  ← 日々のタスク（既存）
  │ - phase_id       │  ★追加
  │ - planned_qty    │
  │ - actual_qty     │
  └─────────┬────────┘
            ↓ 紐づく
【日々】現場作業員
            ↓
  ┌──────────────────┐
  │ photo_reports    │  ← 進捗報告写真
  │ - task_id        │  ★既存
  │ - title          │
  └─────────┬────────┘
            ↓ 自動集計
  ┌──────────────────┐
  │ phase_progress   │  ← 工程進捗（計算結果）
  │ (VIEW)           │
  └─────────┬────────┘
            ↓ 表示
   工程管理画面（ガント風）
   全体進捗バー
   報告書の自動入力データ
```

### 3.3 DBスキーマ

```sql
-- 工程の大区分（既存tasksの親）
CREATE TABLE project_phases (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  site_id UUID NOT NULL REFERENCES sites(id) ON DELETE CASCADE,
  
  -- 識別
  name TEXT NOT NULL,                       -- 「基礎工事」「躯体工事」「内装」
  display_order INTEGER DEFAULT 0,
  color TEXT,                               -- ガント表示用
  
  -- 計画
  planned_start_date DATE,
  planned_end_date DATE,
  
  -- 実績（自動計算）
  actual_start_date DATE,                   -- 最初のタスクが着手された日
  actual_end_date DATE,                     -- 全タスクが完了した日
  
  -- 進捗（自動計算 or 手動上書き）
  progress_pct INTEGER DEFAULT 0,
  progress_mode TEXT DEFAULT 'auto',        -- auto/manual
  
  -- メタ
  notes TEXT,
  created_by UUID REFERENCES users(id),
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_phases_site_order ON project_phases(site_id, display_order);

-- 既存 tasks に phase_id を追加
ALTER TABLE tasks ADD COLUMN phase_id UUID REFERENCES project_phases(id);
ALTER TABLE tasks ADD COLUMN auto_progress_source TEXT DEFAULT 'manual';
  -- manual: 数値手動入力
  -- photo_count: 進捗写真の枚数で自動計算
  -- daily_status: 日々の状態選択で自動計算

-- 進捗集計VIEW
CREATE OR REPLACE VIEW phase_progress_summary AS
SELECT 
  p.id AS phase_id,
  p.site_id,
  p.name AS phase_name,
  p.planned_start_date,
  p.planned_end_date,
  COUNT(t.id) AS total_tasks,
  COUNT(t.id) FILTER (WHERE t.status = 'completed') AS completed_tasks,
  COUNT(t.id) FILTER (WHERE t.status = 'in_progress') AS in_progress_tasks,
  COUNT(t.id) FILTER (WHERE t.status = 'paused') AS paused_tasks,
  CASE 
    WHEN COUNT(t.id) = 0 THEN 0
    WHEN p.progress_mode = 'manual' THEN p.progress_pct
    ELSE ROUND(
      SUM(COALESCE(t.actual_qty, 0)) / NULLIF(SUM(t.planned_qty), 0) * 100
    )::INTEGER
  END AS calculated_progress
FROM project_phases p
LEFT JOIN tasks t ON t.phase_id = p.id
GROUP BY p.id, p.site_id, p.name, p.planned_start_date, p.planned_end_date, 
         p.progress_mode, p.progress_pct;
```

### 3.4 自動進捗反映ロジック

```typescript
// src/lib/progress/auto-update.ts

/**
 * 進捗報告写真が投稿された時に呼ばれる。
 * task の actual_qty を進捗報告から推定して更新。
 */
export async function updateTaskProgressFromPhotoReport(
  photoReportId: string
) {
  const supabase = createServerClient();
  
  // 1. 写真報告を取得
  const { data: photo } = await supabase
    .from('photo_reports')
    .select('*, tasks(*)')
    .eq('id', photoReportId)
    .single();
  
  if (!photo || !photo.task_id || !photo.tasks) return;
  
  const task = photo.tasks;
  
  // 2. autoモードのみ自動更新
  if (task.auto_progress_source === 'manual') return;
  
  if (task.auto_progress_source === 'photo_count') {
    // 進捗写真の枚数 / 予定枚数 で計算
    const { count } = await supabase
      .from('photo_reports')
      .select('id', { count: 'exact' })
      .eq('task_id', task.id)
      .eq('category', 'progress');
    
    const newActualQty = Math.min(count ?? 0, task.planned_qty);
    
    await supabase
      .from('tasks')
      .update({ actual_qty: newActualQty })
      .eq('id', task.id);
  }
  
  // 3. phase の進捗も自動再計算（VIEW経由なので自動）
  
  // 4. site の overall_progress も更新（半自動）
  await refreshSiteProgress(task.site_id);
}

/**
 * 現場全体の進捗を、phase進捗から再計算。
 * ただし監督が手動で進捗を入れている場合は上書きしない。
 */
export async function refreshSiteProgress(siteId: string) {
  const supabase = createServerClient();
  
  const { data: site } = await supabase
    .from('sites')
    .select('progress_mode, progress_pct')
    .eq('id', siteId)
    .single();
  
  if (site?.progress_mode === 'manual') return; // 手動モードは触らない
  
  // 全phaseの平均（重み付けは将来拡張）
  const { data: phases } = await supabase
    .from('phase_progress_summary')
    .select('calculated_progress')
    .eq('site_id', siteId);
  
  if (!phases || phases.length === 0) return;
  
  const avg = Math.round(
    phases.reduce((sum, p) => sum + (p.calculated_progress ?? 0), 0) / phases.length
  );
  
  await supabase
    .from('sites')
    .update({ overall_progress: avg })
    .eq('id', siteId);
}
```

```sql
-- 取引先（発注元）★既存をベースに拡張
CREATE TABLE clients (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  company_id UUID NOT NULL REFERENCES companies(id) ON DELETE CASCADE,
  
  -- 識別
  name TEXT NOT NULL,                       -- 「YKK AP株式会社」
  short_name TEXT,                          -- 「YKK」
  
  -- 取引先タイプ
  client_type TEXT NOT NULL,                -- general_contractor / dispatcher / direct / agent
    -- general_contractor: 元請業者（YKK APなど）
    -- dispatcher: 人材手配会社（パソナなど）
    -- direct: 直接受注（自社元請）
    -- agent: 仲介・代理店

  contact_phone TEXT,
  address TEXT,

  default_report_template_id UUID REFERENCES report_templates(id),

  notes TEXT,
  is_active BOOLEAN DEFAULT TRUE,

-- 施主（最終顧客）★新設
CREATE TABLE end_customers (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  company_id UUID NOT NULL REFERENCES companies(id) ON DELETE CASCADE,
  
  -- 識別
  name TEXT NOT NULL,                       -- 「山本様」「株式会社○○」
  customer_type TEXT,                       -- individual / corporate
  
  -- 連絡先
  contact_person TEXT,                      -- 法人の場合の担当者
  contact_email TEXT,
  contact_phone TEXT,
  postal_code TEXT,
  address TEXT,
  
  -- 工事関連
  total_projects_count INTEGER DEFAULT 0,   -- 過去案件数（自動計算）

  notes TEXT,

CREATE TABLE project_clients (
  project_id UUID NOT NULL REFERENCES projects(id) ON DELETE CASCADE,
  client_id UUID NOT NULL REFERENCES clients(id) ON DELETE CASCADE,
```
