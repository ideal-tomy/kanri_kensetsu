```typescript
  const user = await requireSession();

  }
  return user;
```
```typescript
}
```

#### T2-03：Supabase クライアント抽象化 🟠

```typescript
// src/lib/db/index.ts
import { createServerClient } from '@/lib/supabase/server';

/**
 * 既存の getPhotoReportsForUser と同じシグネチャで Supabase 版を実装。
 * 切替は import 元の変更だけで済むようにする。
 */
export async function getPhotoReportsForUser(user: SessionUser): Promise<PhotoReport[]> {
  const supabase = createServerClient();
  
  if (user.role === 'admin' || user.role === 'owner') {
    const { data } = await supabase
      .from('photo_reports')
      .select('*, users(name), sites(name)')
      .order('created_at', { ascending: false });
    return data?.map(toPhotoReport) ?? [];
  }
  
  if (user.role === 'supervisor') {
    const { data } = await supabase
      .from('photo_reports')
      .select('*, users(name), sites!inner(*)')
      .or(`sites.supervisor_id.eq.${user.id}`)
      .order('created_at', { ascending: false });
    return data?.map(toPhotoReport) ?? [];
  }
  
  // worker
  const { data } = await supabase
    .from('photo_reports')
    .select('*, users(name), sites(name)')
    .eq('user_id', user.id)
    .order('created_at', { ascending: false });
  return data?.map(toPhotoReport) ?? [];
}

function toPhotoReport(row: any): PhotoReport {
  return {
    id: row.id,
    siteId: row.site_id,
    userName: row.users.name,
    category: row.category,
    fileName: row.file_name,
    title: row.title ?? undefined,
    note: row.note ?? undefined,
    storagePath: row.storage_path,
    createdAt: row.created_at,
  };
}
```

**ポイント**：型シグネチャを既存に合わせる → import元の差替えで段階的に移行

#### T2-04：書き込みAPI を shadow write 化 🟠

```typescript
// src/app/api/photos/route.ts
import { state, addPhotoReport } from '@/lib/prototype-store';
import { createServerClient } from '@/lib/supabase/server';

export async function POST(req: Request) {
  const body = await req.json();
  
  // 1. prototype-store（既存）
  const photo = addPhotoReport(body);
  
  // 2. Supabase（新規・shadow write）
  try {
    const supabase = createServerClient();
    await supabase.from('photo_reports').insert({
      id: photo.id,
      site_id: photo.siteId,
      user_id: photo.userId,
      category: photo.category,
      file_name: photo.fileName,
      title: photo.title,
      note: photo.note,
      storage_path: photo.storagePath,
    });
  } catch (e) {
    console.error('Supabase shadow write failed:', e);
    // 失敗してもアプリは継続（prototype-store側は成功している）
  }
  
  return Response.json(photo);
```

## 7. このPLANの背景にある判断

### なぜ Phase を 3 つに分けたか

1. **商談を止めない**：Phase 1 だけでも「データが繋がってる感」が出せる
2. **データロスト回避**：shadow write で段階移行
3. **デモ価値の保全**：AI機能などはmockで残す方が商談で映える

### なぜ Supabase 移行を Phase 2 にしたか

- prototype-store はメモリベース。再起動でデータ消失する
- パイロット導入時に「投稿が消えた」事故を起こせない
- Phase 1 で整合性を整えてから移行する方が事故が少ない

### なぜデモmockを廃止しないか

- AI配員・スキルタグ等は**営業上の差別化要素**
- 「現場ポケット＋AI」の世界観で商談する価値がある
- 「将来実装予定」の明示で誠実さも担保
